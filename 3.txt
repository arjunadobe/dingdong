@echo off
setlocal enabledelayedexpansion

echo ===========================================
echo MAGENTO FILE DIFFERENCE FINDER
echo ===========================================
echo.

set svn=D:\Svn\dev-upgrade\app\code
set other=D:\Chandan\Production_Exisintng_backup_04nov25\app\code

set diff_report=files_with_differences_%date:/=-%.txt

echo Report: %diff_report%
echo.

echo FILES WITH DIFFERENCES > "%diff_report%"
echo ====================== >> "%diff_report%"
echo Generated: %date% %time% >> "%diff_report%"
echo SVN: %svn% >> "%diff_report%"
echo Other: %other% >> "%diff_report%"
echo. >> "%diff_report%"

set diff_count=0
set missing_count=0
set extra_count=0

echo Checking for differences...
echo.

:: Check all file types
for %%e in (php phtml xml js css less scss json txt config) do (
    echo Checking .%%e files...
    
    for /r "%svn%" %%f in (*.%%e) do (
        set "filepath=%%f"
        set "relpath=!filepath:%svn%=!"
        
        if exist "%other%!relpath!" (
            fc "%%f" "%other%!relpath!" > nul
            if errorlevel 1 (
                set /a diff_count+=1
                echo !relpath! >> "%diff_report%"
            )
        ) else (
            set /a missing_count+=1
        )
    )
    
    for /r "%other%" %%f in (*.%%e) do (
        set "filepath=%%f"
        set "relpath=!filepath:%other%=!"
        
        if not exist "%svn%!relpath!" (
            set /a extra_count+=1
        )
    )
)

:: Add summary to report
echo. >> "%diff_report%"
echo ====================== >> "%diff_report%"
echo SUMMARY >> "%diff_report%"
echo ====================== >> "%diff_report%"
echo Files with differences: %diff_count% >> "%diff_report%"
echo Files missing in other: %missing_count% >> "%diff_report%"
echo Extra files in other: %extra_count% >> "%diff_report%"
echo Total files checked: .php .phtml .xml .js .css .less .scss .json .txt .config >> "%diff_report%"

echo.
echo ===========================================
echo COMPLETE
echo ===========================================
echo Files with differences: %diff_count%
echo Missing in other: %missing_count%
echo Extra in other: %extra_count%
echo.
echo Report: %diff_report%
echo.
notepad "%diff_report%"
pause
