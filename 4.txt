@echo off
setlocal enabledelayedexpansion

echo ===========================================
echo MAGENTO FOLDER COMPARISON - HTML WITH DIFFS
echo ===========================================
echo.

set svn=D:\Svn\dev-upgrade\app\code
set other=D:\Chandan\Production_Exisintng_backup_04nov25\app\code

set report=report_with_diffs.html

:: Remove old report
if exist "%report%" del "%report%"

echo Creating HTML report with diffs: %report%
echo.

:: Create HTML header - FIXED PARENTHESES ISSUE
(
echo ^<html^>
echo ^<head^>
echo   ^<title^>Magento File Differences with Details^</title^>
echo   ^<style^>
echo     body { font-family: Arial, sans-serif; margin: 20px; }
echo     h1 { color: #333; }
echo     h2 { color: #555; margin-top: 30px; }
echo     .diff-section { background: #f9f9f9; padding: 15px; margin: 20px 0; border-left: 4px solid #ff6b6b; }
echo     .file-header { background: #e9e9e9; padding: 10px; font-weight: bold; }
echo     .diff-content { background: white; padding: 15px; font-family: Consolas, monospace; font-size: 12px; }
echo     .diff-line { margin: 2px 0; }
echo     .diff-old { color: red; background: #ffe6e6; }
echo     .diff-new { color: green; background: #e6ffe6; }
echo     .summary { background: #f0f8ff; padding: 15px; margin: 20px 0; }
echo     table { border-collapse: collapse; width: 100%%; }
echo     th, td { border: 1px solid #ddd; padding: 8px; }
echo     th { background: #f2f2f2; }
echo   ^</style^>
echo ^</head^>
echo ^<body^>
echo   ^<h1^>Magento File Differences with Details^</h1^>
echo   ^<div class="summary"^>
echo     ^<p^>^<strong^>Generated:^</strong^ %date% %time%^</p^>
echo     ^<p^>^<strong^>SVN Folder:^</strong^ %svn%^</p^>
echo     ^<p^>^<strong^>Other Folder:^</strong^ %other%^</p^>
echo   ^</div^>
) > "%report%"

set diff_count=0

:: Start files with differences section
echo   ^<h2^>Files with Detailed Differences^</h2^> >> "%report%"

:: Check all Magento file types including .less
for %%t in (php phtml xml less css js) do (
    echo Checking .%%t files...
    
    for /r "%svn%" %%f in (*.%%t) do (
        set "filepath=%%f"
        set "relpath=!filepath:%svn%=!"
        
        if exist "%other%!relpath!" (
            :: Create temp file for fc output
            fc "%%f" "%other%!relpath!" > temp_fc.txt
            
            :: Check if different
            if errorlevel 1 (
                set /a diff_count+=1
                
                :: Add to HTML report with diffs - FIXED PARENTHESES
                echo   ^<div class="diff-section"^> >> "%report%"
                echo     ^<div class="file-header"^>File: !relpath! ^(.%%t^) >> "%report%"
                echo ^</div^> >> "%report%"
                echo     ^<div class="diff-content"^> >> "%report%"
                
                :: Read and format diff output
                set in_diff_block=0
                for /f "tokens=*" %%d in (temp_fc.txt) do (
                    set "line=%%d"
                    
                    :: Skip empty lines and headers
                    if not "!line!"=="" (
                        echo "!line!" | findstr /i "comparing files fc:" > nul
                        if errorlevel 1 (
                            :: Check if it's a diff line
                            echo "!line!" | findstr "^\*\*\*\*\*" > nul
                            if !errorlevel! == 0 (
                                set in_diff_block=1
                                echo       ^<div class="diff-line" style="background:#f0f0f0;padding:5px;font-weight:bold;"^>!line!^</div^> >> "%report%"
                            ) else (
                                if !in_diff_block! == 1 (
                                    echo       ^<div class="diff-line diff-old"^>!line!^</div^> >> "%report%"
                                ) else (
                                    echo       ^<div class="diff-line diff-new"^>!line!^</div^> >> "%report%"
                                )
                            )
                        )
                    )
                )
                
                echo     ^</div^> >> "%report%"
                echo   ^</div^> >> "%report%"
                echo. >> "%report%"
            )
            
            del temp_fc.txt 2>nul
        )
    )
)

:: Add summary
(
echo   ^<h2^>Summary^</h2^>
echo   ^<div class="summary"^>
echo     ^<p^>Files with differences: %diff_count%^</p^>
echo     ^<p^>File types checked: php, phtml, xml, less, css, js^</p^>
echo     ^<p^>Report: %report%^</p^>
echo   ^</div^>
echo ^</body^>
echo ^</html^>
) >> "%report%"

echo.
echo ===========================================
echo REPORT COMPLETE
echo ===========================================
echo Files with differences: %diff_count%
echo Report with diffs: %report%
echo.
echo Opening HTML report...
start "" "%report%"
pause
