<?php

namespace Lotus\Merchant\Plugin;

use Magento\Eav\Model\Attribute\Data\Select as Subject;

class SelectValidatePlugin
{
    /**
     * BEFORE plugin — called before validateValue() is executed.
     *
     * @param Subject $subject
     * @param mixed $value
     * @return array
     */
    public function beforeValidateValue(Subject $subject, $value): array
    {
        $attribute = $subject->getAttribute();

        if (!$attribute) {
            return [$value];
        }

        // Attributes to skip the "option does not exist" validation
        $skipAttributes = [
            'address_proof',
            'merchant_categories'
        ];

        if (!in_array($attribute->getAttributeCode(), $skipAttributes, true)) {
            return [$value];
        }

        // If value is invalid or removed from source → make it empty
        if (!empty($value)) {
            $source = $attribute->getSource();
            if ($source && !$source->getOptionText($value)) {
                $value = '';
            }
        }

        return [$value];
    }
}
