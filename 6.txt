    </type>

<?php

namespace Lotus\Merchant\Plugin;

use Magento\Eav\Model\Attribute\Data\Multiselect as Subject;

class MultiSelectValidatePlugin {

    /**
     * AFTER plugin — always return true for merchant_secondary_categories
     *
     * @param Subject $subject
     * @param mixed   $result
     * @param mixed   $value
     * @return mixed
     */
    public function afterValidateValue(Subject $subject, $result, $value)
    {
        $attribute = $subject->getAttribute();

        if ($attribute && $attribute->getAttributeCode() === 'merchant_secondary_categories') {
            return true;
        }

        return $result;
    }

}
