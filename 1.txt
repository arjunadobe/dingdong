<?php
namespace Lotus\TranslationFixer\Plugin;

use Lotus\TranslationFixer\Service\TranslationFileCreator;
use Magento\Framework\App\ResponseInterface;
use Magento\Framework\App\State;
use Magento\Framework\Controller\ResultInterface;

class CreateTranslationFilesOnFrontendPlugin
{
    /**
     * @param TranslationFileCreator $translationFileCreator
     * @param State $appState
     */
    public function __construct(
        private readonly TranslationFileCreator $translationFileCreator,
        private readonly State $appState
    ) {
    }

    /**
     * Create translation files after front controller dispatch
     *
     * @param \Magento\Framework\App\FrontControllerInterface $subject
     * @param ResponseInterface|ResultInterface $result
     * @return ResponseInterface|ResultInterface
     */
    public function afterDispatch(\Magento\Framework\App\FrontControllerInterface $subject, ResponseInterface|ResultInterface $result): ResponseInterface|ResultInterface
    {
        // Only run in developer mode to avoid performance impact in production
        try {
            if ($this->appState->getMode() !== State::MODE_DEVELOPER) {
                return $result;
            }

            $areaCode = $this->appState->getAreaCode();

            // Only check for frontend and adminhtml areas
            if (in_array($areaCode, ['frontend', 'adminhtml'])) {
                // Check if translation files exist for the current area
                if (!$this->translationFileCreator->translationFilesExist($areaCode)) {
                    // Create translation files for the current theme in the current area
                    $this->translationFileCreator->createTranslationFilesForCurrentTheme($areaCode);
                }
            }
        } catch (\Exception $e) {
            // Silent fail - don't break the site if translation files can't be created
        }

        return $result;
    }
}
