    /**
     *
     * @param string $segmentId
     * @param int $storeId
     * @param int|null $limit
     * @return string[]
     * @throws \Exception
     */
    public function getNewProducts($segmentId, $storeId, $limit = null)
    {
        $this->storeManager->setCurrentStore($storeId);
        $productLimit = empty($limit) ? $this->pageSize : (int)$limit;

        // Get option id for "Yes" of isNew attribute (e.g., 36)
        $optionId = $this->getAttributeIdFromOptionValue(self::IS_NEW, "Yes");
        if (!$optionId) {
            return $this->noDataInResponse();
        }

        try {
            /** @var \Magento\Catalog\Model\ResourceModel\Product\Collection $collection */
            $collection = $this->productCollectionFactory->create();
            $collection->setStoreId($storeId);
            $collection->addStoreFilter($storeId);
            $collection->addAttributeToSelect('*');

            // Status & visibility
            $collection->addAttributeToFilter(
                'status',
                \Magento\Catalog\Model\Product\Attribute\Source\Status::STATUS_ENABLED
            );
            $collection->setVisibility($this->productVisibility->getVisibleInCatalogIds());

            // Filter: is_new = Yes (option id)
            $collection->addAttributeToFilter(self::IS_NEW, $optionId);

            // Filter: segment_id (if given)
            if (!empty($segmentId)) {
                $collection->addAttributeToFilter('segment_id', $segmentId);
            }

            // whats_new_rank not empty + sort
            $collection->addAttributeToFilter(self::WHATS_NEW_RANK, ['neq' => '']);
            $collection->addAttributeToSort(self::WHATS_NEW_RANK, self::ASC);

            // Pagination
            $collection->setPageSize($productLimit);
            $collection->setCurPage(1);

            $totalRecords = $collection->getSize();

            // Your existing formatter
            $productData = $this->getProductData($collection, $storeId, $segmentId);

            $link = $this->productLinkFactory->create();
            $link->setProducts($productData)
                 ->setCount($totalRecords);

            return [$link];

        } catch (\Exception $e) {
            throw new \Exception(__($e->getMessage()));
        }
    }

    /**
     * Optional: keep this implemented for interface compatibility.
     * If you don't want it at all, remove it from WhatsNewInterface also.
     *
     * @param string $segmentId
     * @param int $storeId
     * @param string|null $sort
     * @param string|null $direction
     * @param string|null $dir
     * @param int|null $page
     * @return string[]
     * @throws \Exception
     */
    public function getAllNewProducts($segmentId, $storeId, $sort = null, $direction = null, $dir = null, $page = null)
    {
        $this->storeManager->setCurrentStore($storeId);

        $optionId = $this->getAttributeIdFromOptionValue(self::IS_NEW, "Yes");
        if (!$optionId) {
            return $this->noDataInResponse();
        }

        // Default pagination values
        $page      = $page ? (int)$page : 1;
        $pageSize  = $this->pageSize;

        try {
            /** @var \Magento\Catalog\Model\ResourceModel\Product\Collection $collection */
            $collection = $this->productCollectionFactory->create();
            $collection->setStoreId($storeId);
            $collection->addStoreFilter($storeId);
            $collection->addAttributeToSelect('*');

            // Status & visibility
            $collection->addAttributeToFilter(
                'status',
                \Magento\Catalog\Model\Product\Attribute\Source\Status::STATUS_ENABLED
            );
            $collection->setVisibility($this->productVisibility->getVisibleInCatalogIds());

            // Filter: is_new = Yes (option id)
            $collection->addAttributeToFilter(self::IS_NEW, $optionId);

            // Filter: segment_id (if given)
            if (!empty($segmentId)) {
                $collection->addAttributeToFilter('segment_id', $segmentId);
            }

            // whats_new_rank not empty
            $collection->addAttributeToFilter(self::WHATS_NEW_RANK, ['neq' => '']);

            // Sorting: use your helper to preserve existing behavior
            $this->getSortedProductCollection(
                $collection,
                $sort,
                $direction,
                $dir,
                self::WHATS_NEW_RANK
            );

            // Pagination
            $collection->setPageSize($pageSize);
            $collection->setCurPage($page);

            // We don't have SearchResultInterface here, but getPagination/formatedOutput
            // likely expects it; if so, you can compute a lightweight object or
            // simply bypass them and use getProductData directly.
            // For now let's keep it simple and inline it:

            $totalRecords = $collection->getSize();
            $productData  = $this->getProductData($collection, $storeId, $segmentId);

            $link = $this->productLinkFactory->create();
            $link->setProducts($productData)
                 ->setCount($totalRecords);

            return [$link];

        } catch (\Exception $e) {
            throw new \Exception(__($e->getMessage()));
        }
    }
