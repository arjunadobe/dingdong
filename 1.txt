namespace Lotus\Catalog\Model\Product;

use Lotus\Catalog\Api\Product\WhatsNewInterface;

class WhatsNewList extends \Lotus\Catalog\Model\AbstractModel implements WhatsNewInterface
{
    protected $responseArr = [];
    public $pageSize = 10;

    /**
     * @var \Magento\Framework\App\RequestInterface
     */
    private $_request;

    /**
     * @var \Lotus\Catalog\Api\Product\Data\CategoryProductListInterfaceFactory
     */
    protected $productLinkFactory;

    /**
     * @var \Magento\Catalog\Api\CategoryRepositoryInterface
     */
    protected $categoryRepository;

    /**
     * @var \Magento\Store\Model\StoreManagerInterface
     */
    protected $storeManager;

    /**
     * @var \Magento\Framework\App\Config\ScopeConfigInterface
     */
    protected $_scopeConfig;

    /**
     * @var \Magento\Search\Api\SearchInterface
     */
    private $search;

    /**
     * @var \Magento\Framework\Api\Search\SearchCriteriaBuilder
     */
    protected $searchCriteriaBuilder;

    /**
     * @var \Magento\Framework\Api\Search\SearchCriteriaBuilderFactory
     */
    protected $searchCriteriaBuilderFactory;

    /**
     * @var \Magento\Framework\Api\FilterBuilder
     */
    protected $filterBuilder;

    /**
     * @var \Magento\Catalog\Model\Product\Attribute\Source\Status
     */
    protected $productStatus;

    /**
     * @var \Magento\Catalog\Model\Product\Visibility
     */
    protected $productVisibility;

    /**
     * @param \Magento\Catalog\Model\Entity\Attribute $attribute
     * @param \Magento\Store\Model\StoreManagerInterface $storeManager
     * @param \Lotus\Catalog\Api\Product\Data\CategoryProductListInterfaceFactory $productLinkFactory
     * @param \Lotus\Offer\Model\Offer $offerCollection
     * @param \Magento\Catalog\Model\Product\Attribute\Source\Status $productStatus
     * @param \Magento\Catalog\Model\Product\Visibility $productVisibility
     * @param \Magento\Framework\App\RequestInterface $request
     * @param \Magento\Catalog\Api\CategoryRepositoryInterface $categoryRepository
     * @param \Magento\Catalog\Model\ResourceModel\Product\CollectionFactory $productCollectionFactory
     * @param \Magento\Framework\App\Config\ScopeConfigInterface $scopeConfig
     * @param \Magento\Framework\Search\Adapter\Mysql\TemporaryStorageFactory $temporaryStorageFactory
     * @param \Magento\Catalog\Model\Layer\Resolver $layerResolver
     * @param \Magento\Catalog\Model\Product $product
     * @param \Magento\Framework\Api\Search\SearchCriteriaBuilder $searchCriteriaBuilder
     * @param \Magento\Search\Api\SearchInterface $search
     * @param \Magento\Framework\Api\FilterBuilder $filterBuilder
     * @param \Magento\Catalog\Model\Config $catalogConfig
     * @param \Magento\Framework\Api\Search\SearchCriteriaBuilderFactory $searchCriteriaBuilderFactory
     */
    public function __construct(
        \Magento\Catalog\Model\Entity\Attribute $attribute,
        \Magento\Store\Model\StoreManagerInterface $storeManager,
        \Lotus\Catalog\Api\Product\Data\CategoryProductListInterfaceFactory $productLinkFactory,
        \Lotus\Offer\Model\Offer $offerCollection,
        \Magento\Catalog\Model\Product\Attribute\Source\Status $productStatus,
        \Magento\Catalog\Model\Product\Visibility $productVisibility,
        \Magento\Framework\App\RequestInterface $request,
        \Magento\Catalog\Api\CategoryRepositoryInterface $categoryRepository,
        \Magento\Catalog\Model\ResourceModel\Product\CollectionFactory $productCollectionFactory,
        \Magento\Framework\App\Config\ScopeConfigInterface $scopeConfig,
        \Magento\Framework\Search\Adapter\Mysql\TemporaryStorageFactory $temporaryStorageFactory,
        \Magento\Catalog\Model\Layer\Resolver $layerResolver,
        \Magento\Catalog\Model\Product $product,
        \Magento\Framework\Api\Search\SearchCriteriaBuilder $searchCriteriaBuilder,
        \Magento\Search\Api\SearchInterface $search,
        \Magento\Framework\Api\FilterBuilder $filterBuilder,
        \Magento\Catalog\Model\Config $catalogConfig,
        \Magento\Framework\Api\Search\SearchCriteriaBuilderFactory $searchCriteriaBuilderFactory
    ) {
        $this->productLinkFactory            = $productLinkFactory;
        $this->offerCollection               = $offerCollection;
        $this->productStatus                 = $productStatus;
        $this->productVisibility             = $productVisibility;
        $this->_request                      = $request;
        $this->search                        = $search;
        $this->storeManager                  = $storeManager;
        $this->_scopeConfig                  = $scopeConfig;
        $this->searchCriteriaBuilder         = $searchCriteriaBuilder;
        $this->searchCriteriaBuilderFactory  = $searchCriteriaBuilderFactory;
        $this->categoryRepository            = $categoryRepository;
        $this->filterBuilder                 = $filterBuilder;

        parent::__construct(
            $attribute,
            $categoryRepository,
            $productLinkFactory,
            $storeManager,

            $productCollectionFactory,
            $scopeConfig,

            $layerResolver,

            $search,
            $filterBuilder,
            $catalogConfig
        );
    }

    /**
     *
     * @param string $segmentId
     * @param int $storeId
     * @param int|null $limit
     * @return string[]
     * @throws \Exception
     */
    public function getNewProducts($segmentId, $storeId, $limit = null)
    {
        $this->storeManager->setCurrentStore($storeId);
        $productLimit = empty($limit) ? $this->pageSize : (int)$limit;

        // Get option id for "Yes" of isNew attribute
        $optionId = $this->getAttributeIdFromOptionValue(self::IS_NEW, "Yes");
        if (!$optionId) {
            return $this->noDataInResponse();
        }

        try {
            // Use a fresh SearchCriteriaBuilder (Search namespace) instance
            $builder = $this->searchCriteriaBuilderFactory->create();

            // Request name – REQUIRED for SearchInterface::search()
            // This must exist in etc/search_request.xml (catalog_view_container is standard)
            $builder->setRequestName('catalog_view_container');

            // Pagination
            $builder->setPageSize($productLimit);
            $builder->setCurrentPage(1);

            // Filter: isNew = Yes
            $isNewFilter = $this->filterBuilder
                ->setField(self::IS_NEW)
                ->setValue($optionId)
                ->setConditionType('eq')
                ->create();
            $builder->addFilter($isNewFilter);

            // Optional: filter by segment_id
            if (!empty($segmentId)) {
                $segmentFilter = $this->filterBuilder
                    ->setField('segment_id')
                    ->setValue($segmentId)
                    ->setConditionType('eq')
                    ->create();
                $builder->addFilter($segmentFilter);
            }

            $searchCriteria = $builder->create();

            $searchResult  = $this->search->search($searchCriteria);
            $totalRecords  = $searchResult->getTotalCount();
            $items         = $searchResult->getItems();

            $collection = $this->getLayerProductCollection($items, $storeId);
            $collection->addAttributeToSort(self::WHATS_NEW_RANK, self::ASC)
                ->addAttributeToFilter(self::WHATS_NEW_RANK, ['neq' => ''])
                ->setPageSize($productLimit);

            $productData = $this->getProductData($collection, $storeId, $segmentId);

            $link = $this->productLinkFactory->create();
            $link->setProducts($productData)
                ->setCount($totalRecords);

            $links[] = $link;

            return $links;
        } catch (\Exception $e) {
            throw new \Exception(__($e->getMessage()));
        }
    }

    /**
     * @param string $segmentId
     * @param int $storeId
     * @param string|null $sort
     * @param string|null $direction
     * @param string|null $dir
     * @param int|null $page
     * @return string[]
     * @throws \Exception
     */
    public function getAllNewProducts($segmentId, $storeId, $sort = null, $direction = null, $dir = null, $page = null)
    {
        $this->storeManager->setCurrentStore($storeId);

        $optionId = $this->getAttributeIdFromOptionValue(self::IS_NEW, "Yes");
        if (!$optionId) {
            return $this->noDataInResponse();
        }

        try {
            $builder = $this->searchCriteriaBuilderFactory->create();

            // Request name for this search too
            $builder->setRequestName('catalog_view_container');

            // Filter: isNew = Yes
            $isNewFilter = $this->filterBuilder
                ->setField(self::IS_NEW)
                ->setValue($optionId)
                ->setConditionType('eq')
                ->create();
            $builder->addFilter($isNewFilter);

            // Optional: filter by segment_id
            if (!empty($segmentId)) {
                $segmentFilter = $this->filterBuilder
                    ->setField('segment_id')
                    ->setValue($segmentId)
                    ->setConditionType('eq')
                    ->create();
                $builder->addFilter($segmentFilter);
            }

            // Optional adapter-level pagination:
            // $builder->setPageSize($this->pageSize);
            // $builder->setCurrentPage($page ?: 1);

            $searchCriteria = $builder->create();

            $searchResult = $this->search->search($searchCriteria);
            $items        = $searchResult->getItems();

            $collection = $this->getLayerProductCollection($items, $storeId);
            $collection->addAttributeToFilter(self::WHATS_NEW_RANK, ['neq' => '']);

            // Sorting helper from AbstractModel
            $this->getSortedProductCollection(
                $collection,
                $sort,
                $direction,
                $dir,
                self::WHATS_NEW_RANK
            );

            // Pagination helper from AbstractModel
            $productCollection = $this->getPagination($collection, $searchResult, $page);

            return $this->formatedOutput($productCollection, $storeId, $searchResult, $segmentId);
        } catch (\Exception $e) {
            throw new \Exception(__($e->getMessage()));
        }
    }
}
