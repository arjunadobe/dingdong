#!/usr/bin/env bash
#
# magento-deploy.sh
# - Asks for Developer or Production mode
# - Sets Magento mode accordingly
# - Runs setup:upgrade
# - Deploys static content
# - Fixes Magento permissions

set -e  # Exit on first error

########################################
# CONFIG – CHANGE THESE AS NEEDED
########################################
MAGENTO_DIR="/var/www/html/magento"    # Magento root
PHP_BIN="php"                          # PHP binary (php, /usr/bin/php, etc.)
LOCALES="en_US"                        # Static deploy locales (space-separated)
JOBS=4                                 # Parallel jobs for static deploy
WEB_USER="www-data"                    # Web server user
WEB_GROUP="www-data"                   # Web server group
########################################

cd "$MAGENTO_DIR"

echo "============================================"
echo "   Magento Deployment Script"
echo "============================================"
echo "Choose Magento mode:"
echo "  1) Developer"
echo "  2) Production"
echo -n "Enter choice [1-2]: "
read MODE_CHOICE

if [ "$MODE_CHOICE" = "1" ]; then
    TARGET_MODE="developer"
elif [ "$MODE_CHOICE" = "2" ]; then
    TARGET_MODE="production"
else
    echo "Invalid choice. Please run the script again and choose 1 or 2."
    exit 1
fi

echo "==> Selected mode: $TARGET_MODE"

########################################
# 1. Set Magento Mode
########################################
if [ "$TARGET_MODE" = "developer" ]; then
    echo "==> Setting Magento to developer mode..."
    $PHP_BIN bin/magento deploy:mode:set developer -s
else
    echo "==> Enabling maintenance mode..."
    $PHP_BIN bin/magento maintenance:enable || true

    echo "==> Setting Magento to production mode..."
    $PHP_BIN bin/magento deploy:mode:set production -s
fi

########################################
# 2. Setup Upgrade
########################################
echo "==> Running setup:upgrade..."
$PHP_BIN bin/magento setup:upgrade

########################################
# 3. Compile & Static Content
########################################
if [ "$TARGET_MODE" = "production" ]; then
    echo "==> Running DI compile (production)..."
    $PHP_BIN bin/magento setup:di:compile
fi

echo "==> Deploying static content..."
$PHP_BIN bin/magento setup:static-content:deploy -f $LOCALES --jobs=$JOBS

########################################
# 4. Permissions
########################################
echo "==> Setting ownership to $WEB_USER:$WEB_GROUP ..."
sudo chown -R $WEB_USER:$WEB_GROUP .

echo "==> Applying write permissions to required directories and files..."
find var generated vendor pub/static pub/media app/etc -type f -exec chmod u+w {} +
find var generated vendor pub/static pub/media app/etc -type d -exec chmod u+w {} +

echo "==> Ensuring bin/magento is executable..."
chmod u+x bin/magento

########################################
# 5. Cache & Maintenance
########################################
echo "==> Flushing cache..."
$PHP_BIN bin/magento cache:flush

if [ "$TARGET_MODE" = "production" ]; then
    echo "==> Disabling maintenance mode..."
    $PHP_BIN bin/magento maintenance:disable
fi

echo "============================================"
echo " Deployment finished in $TARGET_MODE mode."
echo "============================================"
