#!/usr/bin/env bash

set -e

########################################
# CONFIG
########################################
MAGENTO_DIR="/data/www/html/magentolotus"
SOURCE_CODE="/home/lotusadm/app/code"
SOURCE_DESIGN="/home/lotusadm/app/design"

TARGET_CODE="$MAGENTO_DIR/app/code"
TARGET_DESIGN="$MAGENTO_DIR/app/design"

PHP_BIN="php"
LOCALES="en_US"
JOBS=4
WEB_USER="www-data"
WEB_GROUP="www-data"

########################################
# ASK MODE
########################################
echo "============================================"
echo "   Magento Deployment Script"
echo "============================================"
echo "Choose Magento mode:"
echo "  1) Developer"
echo "  2) Production"
echo -n "Enter choice [1-2]: "
read MODE_CHOICE

if [ "$MODE_CHOICE" = "1" ]; then
    TARGET_MODE="developer"
elif [ "$MODE_CHOICE" = "2" ]; then
    TARGET_MODE="production"
else
    echo "Invalid choice. Select 1 or 2."
    exit 1
fi

echo "==> Selected mode: $TARGET_MODE"

########################################
# ASK COPY/REPLACE APP/CODE & APP/DESIGN
########################################
echo -n "Do you want to REMOVE and COPY app/code & app/design from /home/lotusadm? [y/N]: "
read REFRESH_CHOICE

REFRESH_CHOICE_LOWER=$(echo "$REFRESH_CHOICE" | tr 'A-Z' 'a-z')

########################################
# SWITCH DIRECTORY
########################################
cd "$MAGENTO_DIR"

########################################
# OPTIONAL CODE/DESIGN REFRESH (BOTH MODES)
########################################
if [ "$REFRESH_CHOICE_LOWER" = "y" ] || [ "$REFRESH_CHOICE_LOWER" = "yes" ]; then
    echo "==> Removing existing app/code and app/design..."
    rm -rf "$TARGET_CODE"
    rm -rf "$TARGET_DESIGN"

    echo "==> Copying fresh app/code from $SOURCE_CODE ..."
    cp -R "$SOURCE_CODE" "$TARGET_CODE"

    echo "==> Copying fresh app/design from $SOURCE_DESIGN ..."
    cp -R "$SOURCE_DESIGN" "$TARGET_DESIGN"
else
    echo "==> Skipping app/code and app/design refresh."
fi

########################################
# SET MODE
########################################
if [ "$TARGET_MODE" = "developer" ]; then
    echo "==> Setting Magento to developer mode..."
    $PHP_BIN bin/magento deploy:mode:set developer -s
else
    echo "==> Enabling maintenance mode..."
    $PHP_BIN bin/magento maintenance:enable || true

    echo "==> Setting Magento to production mode..."
    $PHP_BIN bin/magento deploy:mode:set production -s
fi

########################################
# SETUP UPGRADE
########################################
echo "==> Running setup:upgrade..."
$PHP_BIN bin/magento setup:upgrade

########################################
# COMPILATION (PRODUCTION ONLY)
########################################
if [ "$TARGET_MODE" = "production" ]; then
    echo "==> Running DI compile..."
    $PHP_BIN bin/magento setup:di:compile
fi

########################################
# STATIC CONTENT DEPLOY
########################################
echo "==> Deploying static content..."
$PHP_BIN bin/magento setup:static-content:deploy -f $LOCALES --jobs=$JOBS

########################################
# PERMISSIONS
########################################
echo "==> Setting ownership to $WEB_USER:$WEB_GROUP ..."
sudo chown -R $WEB_USER:$WEB_GROUP .

echo "==> Applying write permissions..."
find var generated vendor pub/static pub/media app/etc -type f -exec chmod u+w {} +
find var generated vendor pub/static pub/media app/etc -type d -exec chmod u+w {} +

chmod u+x bin/magento

########################################
# CACHE + MAINTENANCE
########################################
echo "==> Flushing cache..."
$PHP_BIN bin/magento cache:flush

if [ "$TARGET_MODE" = "production" ]; then
    echo "==> Disabling maintenance mode..."
    $PHP_BIN bin/magento maintenance:disable
fi

echo "============================================"
echo " Deployment completed in $TARGET_MODE mode."
echo "============================================"
