@echo off
setlocal enabledelayedexpansion

echo ===========================================
echo MAGENTO app/code FOLDER COMPARISON
echo ===========================================
echo.

:: Set your folder paths
set svn_folder=.\magento-svn\app\code
set other_folder=.\magento-other\app\code

:: OR use full paths
:: set svn_folder=C:\projects\magento-svn\app\code
:: set other_folder=D:\backup\magento-other\app\code

echo Comparing: %svn_folder%
echo         vs %other_folder%
echo ===========================================
echo.

:: Initialize counters
set missing_count=0
set extra_count=0
set diff_count=0
set same_count=0

echo STEP 1: Check folder structure
echo -------------------------------------------
dir "%svn_folder%" /ad /b > svn_vendors.txt
dir "%other_folder%" /ad /b > other_vendors.txt

echo Vendor namespace differences:
fc svn_vendors.txt other_vendors.txt
echo.

:: Compare each vendor namespace
for /f "delims=" %%v in (svn_vendors.txt) do (
    echo.
    echo Checking vendor: %%v
    echo -------------------------
    
    :: Check modules in this vendor
    dir "%svn_folder%\%%v" /ad /b > svn_modules_%%v.txt 2>nul
    dir "%other_folder%\%%v" /ad /b > other_modules_%%v.txt 2>nul
    
    fc svn_modules_%%v.txt other_modules_%%v.txt >nul
    if errorlevel 1 (
        echo Module structure differences in %%v:
        fc svn_modules_%%v.txt other_modules_%%v.txt
    ) else (
        echo Module structure: OK
    )
    
    :: Clean up temp files
    del svn_modules_%%v.txt other_modules_%%v.txt 2>nul
)
echo.

echo STEP 2: Compare all PHP files
echo -------------------------------------------
for /r "%svn_folder%" %%f in (*.php) do (
    set "relpath=%%f"
    set "relpath=!relpath:%svn_folder%=!"
    
    if exist "%other_folder%!relpath!" (
        fc "%%f" "%other_folder%!relpath!" > nul
        if errorlevel 1 (
            set /a diff_count+=1
            echo [DIFF] !relpath!
        ) else (
            set /a same_count+=1
        )
    ) else (
        set /a missing_count+=1
        echo [MISSING in other] !relpath!
    )
)
echo.

echo STEP 3: Find extra files in other folder
echo -------------------------------------------
for /r "%other_folder%" %%f in (*.php) do (
    set "relpath=%%f"
    set "relpath=!relpath:%other_folder%=!"
    
    if not exist "%svn_folder%!relpath!" (
        set /a extra_count+=1
        echo [EXTRA in other] !relpath!
    )
)
echo.

echo STEP 4: Compare XML files (config, di, layout, etc.)
echo -------------------------------------------
for /r "%svn_folder%" %%f in (*.xml) do (
    set "relpath=%%f"
    set "relpath=!relpath:%svn_folder%=!"
    
    :: Only check XML files in specific directories
    echo !relpath! | findstr /i "etc\|view\|layout" > nul
    if !errorlevel! == 0 (
        if exist "%other_folder%!relpath!" (
            fc "%%f" "%other_folder%!relpath!" > nul
            if errorlevel 1 (
                set /a diff_count+=1
                echo [XML DIFF] !relpath!
            )
        ) else (
            echo [XML MISSING] !relpath!
        )
    )
)
echo.

echo ===========================================
echo COMPARISON SUMMARY - app/code ONLY
echo ===========================================
echo Files identical: %same_count%
echo Files with differences: %diff_count%
echo Files missing in other folder: %missing_count%
echo Extra files in other folder: %extra_count%
echo.
echo To view specific file differences:
echo   fc "%svn_folder%\Vendor\Module\file.php" "%other_folder%\Vendor\Module\file.php"
echo.
echo ===========================================
pause
