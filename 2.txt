@echo off
setlocal enabledelayedexpansion

echo ===========================================
echo MAGENTO app/code FOLDER COMPARISON
echo HTML Report Generator
echo ===========================================
echo.

:: Set your folder paths
set svn_folder=.\magento-svn\app\code
set other_folder=.\magento-other\app\code

:: OR use full paths
:: set svn_folder=C:\projects\magento-svn\app\code
:: set other_folder=D:\backup\magento-other\app\code

echo Comparing: %svn_folder%
echo         vs %other_folder%
echo ===========================================
echo.

:: Initialize counters
set missing_count=0
set extra_count=0
set diff_count=0
set same_count=0
set total_files=0

:: Create HTML report file with date format: Report_YYYY-MM-DD_HH-MM.html
set year=%date:~10,4%
set month=%date:~4,2%
set day=%date:~7,2%
set hour=%time:~0,2%
set minute=%time:~3,2%
set second=%time:~6,2%

:: Fix hour if single digit
if "!hour!"==" 0" set hour=00
if "!hour!"==" 1" set hour=01
if "!hour!"==" 2" set hour=02
if "!hour!"==" 3" set hour=03
if "!hour!"==" 4" set hour=04
if "!hour!"==" 5" set hour=05
if "!hour!"==" 6" set hour=06
if "!hour!"==" 7" set hour=07
if "!hour!"==" 8" set hour=08
if "!hour!"==" 9" set hour=09
if "!hour!"=="10" set hour=10
if "!hour!"=="11" set hour=11

set html_file=Magento_Comparison_%year%-%month%-%day%_%hour%-%min%-%second%.html

echo Generating HTML report: %html_file%
echo.

:: Start HTML report
(
echo ^<html^>
echo ^<head^>
echo   ^<title^>Magento Folder Comparison Report^</title^>
echo   ^<style^>

echo     .extra { color: #5bc0de; background-color: #d9edf7; padding: 2px 5px; }
echo     .same { color: #5cb85c; background-color: #dff0d8; padding: 2px 5px; }
echo     table { border-collapse: collapse; width: 100^%; margin-top: 20px; }
echo     th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
echo     th { background-color: #4CAF50; color: white; }
echo     tr:nth-child(even) { background-color: #f2f2f2; }
echo     .timestamp { color: #777; font-size: 12px; }
echo     .file-count { font-weight: bold; color: #333; }
echo   ^</style^>
echo ^</head^>
echo ^<body^>
echo   ^<h1^>^<img src="data:image/png;base64,iVBORw0KGgoAAAANSUhEUgAAABAAAAAQCAYAAAAf8/9hAAAACXBIWXMAAAsTAAALEwEAmpwYAAAAAXNSR0IArs4c6QAAAARnQU1BAACxjwv8YQUAAAAgSURBVHgB7cxBEQAACAIwqPx/zwMVFpZqk1VrD+3eAe9rA1O1ARmzAAAAAElFTkSuQmCC" style="vertical-align: middle; margin-right: 10px;"^>Magento Folder Comparison Report^</h1^>
echo   ^<div class="timestamp"^>Report generated: %date% %time%^</div^>
) > "%html_file%"

:: Add folder info
(
echo   ^<div class="summary"^>
echo     ^<h2^>Comparison Details^</h2^>
echo     ^<p^>^<strong^>SVN Folder:^</strong^^<br^>%svn_folder%^</p^>
echo     ^<p^>^<strong^>Other Folder:^</strong^><br^>%other_folder%^</p^>
echo   ^</div^>
) >> "%html_file%"

:: Create arrays for different file types
set file_extensions=php phtml xml js css less scss json txt config dist md sql

echo Starting comparison... This may take a moment.
echo.

:: STEP 1: Compare all file types
(
echo   ^<h2^>File Differences^</h2^>
echo   ^<table^>
echo     ^<tr^>
echo       ^<th^>Status^</th^>
echo       ^<th^>File Type^</th^>
echo       ^<th^>File Path^</th^>
echo       ^<th^>Action^</th^>
echo     ^</tr^>
) >> "%html_file%"

for %%e in (%file_extensions%) do (
    echo Checking .%%e files...
    
    for /r "%svn_folder%" %%f in (*.%%e) do (
        set /a total_files+=1
        set "relpath=%%f"
        set "relpath=!relpath:%svn_folder%=!"
        set "filename=%%~nxf"
        
        if exist "%other_folder%!relpath!" (
            fc "%%f" "%other_folder%!relpath!" > nul
            if errorlevel 1 (
                set /a diff_count+=1
                (
                echo     ^<tr^>
                echo       ^<td^>^<span class="diff"^>DIFFERENT^</span^>^</td^>
                echo       ^<td^>.%%e^</td^>
                echo       ^<td^>!relpath!^</td^>
                echo       ^<td^>^<a href="javascript:void(0)" onclick="showDiff('%%e', '!relpath!')"^>View Diff^</a^>^</td^>
                echo     ^</tr^>
                ) >> "%html_file%"
            ) else (
                set /a same_count+=1
                (
                echo     ^<tr^>
                echo       ^<td^>^<span class="same"^>IDENTICAL^</span^>^</td^>
                echo       ^<td^>.%%e^</td^>
                echo       ^<td^>!relpath!^</td^>
                echo       ^<td^>^<a href="javascript:void(0)" onclick="viewFile('%%e', '!relpath!', 'svn')"^>View^</a^>^</td^>
                echo     ^</tr^>
                ) >> "%html_file%"
            )
        ) else (
            set /a missing_count+=1
            (
            echo     ^<tr^>
            echo       ^<td^>^<span class="missing"^>MISSING^</span^>^</td^>
            echo       ^<td^>.%%e^</td^>
            echo       ^<td^>!relpath!^</td^>
            echo       ^<td^>Only in SVN^</td^>
            echo     ^</tr^>
            ) >> "%html_file%"
        )
    )
)

:: STEP 2: Find extra files in other folder
for %%e in (%file_extensions%) do (
    for /r "%other_folder%" %%f in (*.%%e) do (
        set "relpath=%%f"
        set "relpath=!relpath:%other_folder%=!"
        
        if not exist "%svn_folder%!relpath!" (
            set /a extra_count+=1
            (
            echo     ^<tr^>
            echo       ^<td^>^<span class="extra"^>EXTRA^</span^>^</td^>
            echo       ^<td^>.%%e^</td^>
            echo       ^<td^>!relpath!^</td^>
            echo       ^<td^>Only in Other^</td^>
            echo     ^</tr^>
            ) >> "%html_file%"
        )
    )
)

(
echo   ^</table^>
) >> "%html_file%"

:: STEP 3: Add summary section
(
echo   ^<h2^>Comparison Summary^</h2^>
echo   ^<div class="summary"^>
echo     ^<table^>
echo       ^<tr^>^<td^>Total Files Checked:^</td^>^<td class="file-count"^>%total_files%^</td^>^</tr^>
echo       ^<tr^>^<td^>Identical Files:^</td^>^<td class="same"^>%same_count%^</td^>^</tr^>
echo       ^<tr^>^<td^>Different Files:^</td^>^<td class="diff"^>%diff_count%^</td^>^</tr^>
echo       ^<tr^>^<td^>Missing in Other:^</td^>^<td class="missing"^>%missing_count%^</td^>^</tr^>
echo       ^<tr^>^<td^>Extra in Other:^</td^>^<td class="extra"^>%extra_count%^</td^>^</tr^>
echo       ^<tr^>^<td^>File Types Checked:^</td^>^<td^>%file_extensions%^</td^>^</tr^>
echo     ^</table^>
echo   ^</div^>
) >> "%html_file%"

:: STEP 4: Add JavaScript for interactivity
(
echo   ^<h2^>Quick Actions^</h2^>
echo   ^<div class="summary"^>
echo     ^<button onclick="expandAll()"^>Expand All Differences^</button^>
echo     ^<button onclick="collapseAll()"^>Collapse All^</button^>
echo     ^<button onclick="exportToCSV()"^>Export to CSV^</button^>
echo     ^<button onclick="window.print()"^>Print Report^</button^>
echo   ^</div^>

echo   ^<h2^>Statistics by File Type^</h2^>
echo   ^<div id="chartContainer" style="height: 300px; width: 100^%;"^>^</div^>

echo   ^<script^>
echo     // Function to show file differences
echo     function showDiff(fileType, filePath) {
echo       alert('Showing differences for: ' + filePath + '\nFile type: .' + fileType);
echo       // In a real implementation, you would fetch and display the actual diff
echo     }
echo     
echo     // Function to view file content

echo     
echo     // Expand/Collapse functions

echo     
echo     // Export to CSV
echo     function exportToCSV() {
echo       var csv = [];

echo       
echo       for (var i = 0; i < rows.length; i++) {
echo         var row = [], cols = rows[i].querySelectorAll('td, th');
echo         
echo         for (var j = 0; j < cols.length; j++) {
echo           row.push(cols[j].innerText);
echo         }
echo         
echo         csv.push(row.join(','));
echo       }
echo       
echo       var csvFile = new Blob([csv.join('\\n')], {type: 'text/csv'});
echo       var downloadLink = document.createElement('a');
echo       downloadLink.download = 'Magento_Comparison_%year%-%month%-%day%.csv';
echo       downloadLink.href = window.URL.createObjectURL(csvFile);
echo       downloadLink.style.display = 'none';

echo     }
echo     
echo     // Initialize page

echo       console.log('Different files: %diff_count%');
echo     });
echo   ^</script^>

echo   ^<footer style="margin-top: 50px; padding-top: 20px; border-top: 1px solid #ccc; color: #777;"^>

echo     ^<p^>File: %html_file%^</p^>

echo   ^</footer^>
echo ^</body^>
echo ^</html^>
) >> "%html_file%"
