@echo off
setlocal enabledelayedexpansion

echo ===========================================
echo MAGENTO FOLDER COMPARISON WITH DIFF DETAILS
echo ===========================================
echo.

set svn=D:\Svn\dev-upgrade\app\code
set other=D:\Chandan\Production_Exisintng_backup_04nov25\app\code

set report=detailed_diff_%date:/=-%.txt

echo Creating detailed report: %report%
echo.

echo DETAILED FILE COMPARISON REPORT > "%report%"
echo =============================== >> "%report%"
echo Date: %date% %time% >> "%report%"
echo SVN Folder: %svn% >> "%report%"
echo Other Folder: %other% >> "%report%"
echo. >> "%report%"

set diff_count=0

echo Checking files for differences...
echo.

:: Check PHP files
for /r "%svn%" %%f in (*.php) do (
    set "filepath=%%f"
    set "relpath=!filepath:%svn%=!"
    
    if exist "%other%!relpath!" (
        :: Run fc and capture output
        fc "%%f" "%other%!relpath!" > temp_fc.txt
        
        :: Check if files are different (fc returns errorlevel 1 if different)
        if errorlevel 1 (
            set /a diff_count+=1
            echo ========================================== >> "%report%"
            echo FILE: !relpath! >> "%report%"
            echo ========================================== >> "%report%"
            
            :: Show first 20 lines of differences
            set line_num=0
            for /f "tokens=*" %%d in (temp_fc.txt) do (
                set /a line_num+=1
                if !line_num! LEQ 30 (
                    echo %%d >> "%report%"
                )
            )
            echo. >> "%report%"
            echo [Showing first 30 lines of differences] >> "%report%"
            echo. >> "%report%"
        )
        del temp_fc.txt 2>nul
    )
)

:: Add summary
echo. >> "%report%"
echo ========================================== >> "%report%"
echo SUMMARY >> "%report%"
echo ========================================== >> "%report%"
echo Total different files found: %diff_count% >> "%report%"
echo Report saved to: %report% >> "%report%"

echo.
echo ===========================================
echo REPORT COMPLETE
echo ===========================================
echo Different files found: %diff_count%
echo Detailed report: %report%
echo.
echo Opening report...
notepad "%report%"
pause
