function testLdapConnection($server, $username, $password) {
    echo "   Connecting to: $server\n";
    echo "   Username: $username\n";

    $ldapConn = @ldap_connect($server, 389);
    if (!$ldapConn) {
        return [false, "Connection failed"];
    }

    // Set LDAP options
    ldap_set_option($ldapConn, LDAP_OPT_PROTOCOL_VERSION, 3);
    ldap_set_option($ldapConn, LDAP_OPT_REFERRALS, 0);
    ldap_set_option($ldapConn, LDAP_OPT_NETWORK_TIMEOUT, 10);
    ldap_set_option($ldapConn, LDAP_OPT_TIMELIMIT, 10);

    // Attempt bind
    $bindResult = @ldap_bind($ldapConn, $username, $password);

    if ($bindResult) {
        return [true, $ldapConn, "Bind successful"];
    } else {
        $error = ldap_error($ldapConn);
        $errno = ldap_errno($ldapConn);
        ldap_close($ldapConn);
        return [false, "Bind failed: $error (Code: $errno)"];
    }
}

function searchLdap($ldapConn, $baseDn, $filter, $attributes = []) {
    $searchResult = @ldap_search($ldapConn, $baseDn, $filter, $attributes, 0, 0, 10);

    if (!$searchResult) {
        return [false, "Search failed: " . ldap_error($ldapConn)];
    }

    $entries = ldap_get_entries($ldapConn, $searchResult);
    ldap_free_result($searchResult);

    return [true, $entries];
}

function discoverBaseDn($ldapConn, $server) {
    // Try common base DNs
    $commonBaseDns = [
        "OU=Users,OU=LOTUS,OU=Others Location,DC=CORP,DC=AD,DC=SBI",
        "OU=Users,OU=LOTUS,DC=CORP,DC=AD,DC=SBI",
        "OU=LOTUS,DC=CORP,DC=AD,DC=SBI",
        "DC=CORP,DC=AD,DC=SBI",
        "CN=Users,DC=CORP,DC=AD,DC=SBI"
    ];

    foreach ($commonBaseDns as $testDn) {
        $searchResult = @ldap_search($ldapConn, $testDn, "(objectClass=*)", [], 0, 0, 5);
        if ($searchResult) {
            $entries = ldap_get_entries($ldapConn, $searchResult);
            if ($entries['count'] > 0) {
                echo "   ✅ Found valid Base DN: $testDn\n";
                ldap_free_result($searchResult);
                return;
            }
            ldap_free_result($searchResult);
        }
    }
    echo "   ❌ Could not discover Base DN automatically\n";
}

function tryAlternativeSearches($ldapConn, $baseDn, $username) {
    $filters = [
        "(cn=$username*)",
        "(userprincipalname=*$username*)",
        "(mail=*$username*)",
        "(displayname=*$username*)"
    ];

    foreach ($filters as $filter) {
        $searchResult = @ldap_search($ldapConn, $baseDn, $filter, ['cn', 'samaccountname'], 0, 1, 5);
        if ($searchResult) {
            $entries = ldap_get_entries($ldapConn, $searchResult);
            if ($entries['count'] > 0) {
                echo "   🔍 Found similar user with filter: $filter\n";
                displayUserInfo($entries[0]);
                ldap_free_result($searchResult);
                return;
            }
            ldap_free_result($searchResult);
        }
    }
    echo "   ℹ️  No similar users found with alternative filters\n";
}

function displayUserInfo($userEntry) {
    $attributes = [
        'dn' => 'Distinguished Name',
        'cn' => 'Common Name',
        'samaccountname' => 'SAM Account',
        'mail' => 'Email',
        'userprincipalname' => 'UPN',
        'displayname' => 'Display Name'
    ];

    foreach ($attributes as $key => $label) {
        if (isset($userEntry[$key])) {
            if (is_array($userEntry[$key])) {
                echo "     $label: " . $userEntry[$key][0] . "\n";
            } else {
                echo "     $label: " . $userEntry[$key] . "\n";
            }
        }
    }
}

function testUserAuthentication($adminLdapConn, $baseDn, $username, $password) {
    echo "\n--- Testing auth for: $username ---\n";

    // First find the user
    $filter = "(samaccountname=$username)";
    $searchResult = @ldap_search($adminLdapConn, $baseDn, $filter, ['dn']);

    if (!$searchResult) {
        echo "   ❌ Search failed: " . ldap_error($adminLdapConn) . "\n";
        return;
    }

    $entries = ldap_get_entries($adminLdapConn, $searchResult);
    ldap_free_result($searchResult);

    if ($entries['count'] == 0) {
        echo "   ❌ User not found\n";
        return;
    }

    $userDn = $entries[0]['dn'];
    echo "   User DN: $userDn\n";

    // Try to bind with user credentials
    $userLdapConn = @ldap_connect($GLOBALS['workingServer'], 389);
    if (!$userLdapConn) {
        echo "   ❌ Connection failed\n";
        return;
    }

    ldap_set_option($userLdapConn, LDAP_OPT_PROTOCOL_VERSION, 3);
    ldap_set_option($userLdapConn, LDAP_OPT_REFERRALS, 0);

    $userBind = @ldap_bind($userLdapConn, $userDn, $password);

    if ($userBind) {
        echo "   ✅ Authentication SUCCESSFUL\n";

        // Get user info with their own credentials
        $userSearch = @ldap_read($userLdapConn, $userDn, "(objectClass=*)", ['cn', 'mail']);
        if ($userSearch) {
            $userInfo = ldap_get_entries($userLdapConn, $userSearch);
            if ($userInfo['count'] > 0) {
                echo "   User can access their own data\n";
            }
            ldap_free_result($userSearch);
        }
    } else {
        echo "   ❌ Authentication FAILED: " . ldap_error($userLdapConn) . "\n";
    }

    ldap_close($userLdapConn);
}
